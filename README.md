     ORDER_ENDPOINT: URL de tu webhook / app COD (ReCompy, Releasit o servidor intermedio) que cree el pedido en Shopify.
     No uses /admin/api/orders.json directamente desde el navegador: requiere un token privado que quedaría expuesto.
     Si queda vacío, el pedido se envía por WhatsApp como respaldo. */
  const CONFIG = {
    ORDER_ENDPOINT: '',
    WHATSAPP: '573027350151',
    PRICES: { 1: 79900, 2: 129900 },
    PRODUCT: 'Mini GPS GF-07'
  };

  const $ = (id) => document.getElementById(id);
  const money = (n) => '$' + n.toLocaleString('es-CO');

  // Fecha estimada de entrega (2 a 5 días hábiles)
  function addBusinessDays(date, days) {
    const d = new Date(date);
    while (days > 0) { d.setDate(d.getDate() + 1); if (d.getDay() !== 0 && d.getDay() !== 6) days--; }
    return d;
  }
  const fmt = new Intl.DateTimeFormat('es-CO', { day: 'numeric', month: 'short' });
  const today = new Date();
  $('eta').textContent = `${fmt.format(addBusinessDays(today, 2))} – ${fmt.format(addBusinessDays(today, 5))}`;

  // Departamentos
  ['Amazonas','Antioquia','Arauca','Atlántico','Bogotá D.C.','Bolívar','Boyacá','Caldas','Caquetá','Casanare','Cauca','Cesar','Chocó','Córdoba','Cundinamarca','Guainía','Guaviare','Huila','La Guajira','Magdalena','Meta','Nariño','Norte de Santander','Putumayo','Quindío','Risaralda','San Andrés y Providencia','Santander','Sucre','Tolima','Valle del Cauca','Vaupés','Vichada']
    .forEach((d) => $('departamento').add(new Option(d, d)));

  // Total dinámico y preselección desde los paquetes
  const qty = $('cantidad');
  const updateTotal = () => { $('total').textContent = money(CONFIG.PRICES[qty.value]); };
  qty.addEventListener('change', updateTotal);
  document.querySelectorAll('[data-qty]').forEach((a) => a.addEventListener('click', () => { qty.value = a.dataset.qty; updateTotal(); }));
  updateTotal();

  // Animación de aparición
  const reveals = document.querySelectorAll('.reveal');
  if ('IntersectionObserver' in window) {
    const io = new IntersectionObserver((es) => es.forEach((e) => { if (e.isIntersecting) { e.target.classList.add('in'); io.unobserve(e.target); } }), { threshold: .1 });
    reveals.forEach((el) => io.observe(el));
  } else reveals.forEach((el) => el.classList.add('in'));

  // Validación
  const form = $('order-form');
  const rules = {
    nombre: (v) => v.trim().split(/\s+/).length >= 2 && v.trim().length >= 5 || 'Escribe tu nombre y apellidos completos.',
    celular: (v) => /^3\d{9}$/.test(v.replace(/[\s-]/g, '')) || 'Ingresa un celular colombiano de 10 dígitos (ej. 3001234567).',
    departamento: (v) => !!v || 'Selecciona tu departamento.',
    ciudad: (v) => v.trim().length >= 2 || 'Indica tu municipio o ciudad.',
    direccion: (v) => v.trim().length >= 8 || 'Escribe una dirección detallada.',
    referencia: (v) => v.trim().length >= 3 || 'Indica un punto de referencia.'
  };
  function setError(el, msg) {
    const wrap = el.closest('div');
    wrap.querySelector('.err')?.remove();
    el.classList.toggle('invalid', !!msg);
    el.setAttribute('aria-invalid', msg ? 'true' : 'false');
    if (msg) { const p = document.createElement('p'); p.className = 'err'; p.textContent = msg; el.insertAdjacentElement('afterend', p); }
  }
  function check(el) {
    const r = rules[el.name]; if (!r) return true;
    const res = r(el.value); setError(el, res === true ? '' : res); return res === true;
  }
  Object.keys(rules).forEach((n) => {
    const el = form.elements[n];
    el.addEventListener('blur', () => check(el));
    el.addEventListener('input', () => el.classList.contains('invalid') && check(el));
  });

  const errBox = $('form-error');
  const showErr = (m) => { errBox.hidden = !m; errBox.className = 'span2 form-error'; errBox.textContent = m || ''; };

  form.addEventListener('submit', async (e) => {
    e.preventDefault();
    showErr('');
    const fields = Object.keys(rules).map((n) => form.elements[n]);
    const ok = fields.map(check).every(Boolean);
    if (!$('terminos').checked) { showErr('Debes aceptar los Términos y las Políticas para continuar.'); return; }
    if (!ok) { fields.find((f) => f.classList.contains('invalid')).focus(); return; }

    const btn = $('submit-btn');
    if (btn.disabled) return;
    btn.disabled = true; btn.textContent = 'Procesando pedido...';

    const q = Number(qty.value);
    const order = {
      producto: CONFIG.PRODUCT,
      cantidad: q,
      total: CONFIG.PRICES[q],
      metodo_pago: 'Pago contra entrega en efectivo',
      nombre: form.nombre.value.trim(),
      celular: form.celular.value.replace(/[\s-]/g, ''),
      departamento: form.departamento.value,
      ciudad: form.ciudad.value.trim(),
      direccion: form.direccion.value.trim(),
      referencia: form.referencia.value.trim(),
      origen: location.href
    };

    try {
      if (CONFIG.ORDER_ENDPOINT) {
        const res = await fetch(CONFIG.ORDER_ENDPOINT, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(order) });
        if (!res.ok) throw new Error('HTTP ' + res.status);
        $('success-msg').textContent = `Gracias, ${order.nombre.split(' ')[0]}. Te enviaremos la confirmación y el número de guía por WhatsApp al ${order.celular}.`;
      } else {
        const text = `Hola AoneClicks Shopping, quiero mi pedido:%0A${encodeURIComponent(
`Producto: ${order.producto} x${order.cantidad}
Total a pagar al recibir: ${money(order.total)}
Nombre: ${order.nombre}
Celular: ${order.celular}
Ubicación: ${order.ciudad}, ${order.departamento}
Dirección: ${order.direccion}
Referencia: ${order.referencia}`)}`;
        window.open(`https://wa.me/${CONFIG.WHATSAPP}?text=${text}`, '_blank', 'noopener');
        $('success-msg').textContent = 'Se abrió WhatsApp con el resumen de tu pedido. Envía el mensaje para que un asesor lo confirme.';
      }
      form.hidden = true;
      const ok = $('success'); ok.hidden = false; ok.focus(); ok.scrollIntoView({ behavior: 'smooth', block: 'center' });
    } catch (err) {
      showErr('No pudimos registrar tu pedido. Inténtalo de nuevo o escríbenos por WhatsApp al +57 302 7350151.');
      btn.disabled = false; btn.textContent = 'PEDIR AHORA Y PAGAR AL RECIBIR';
    }
  });
})();
</script>
</body>
</html>
